# Aliases

Arrow (AR) - Left, Right, Down, Up
Direction (DR) - h, j, k, l

# Common

:noh - disable search
A-Arrow - buffer sizing
<leader>mDirection - move buffer
<leader>sp - split screen
<leader>ss - toggle spell check
<leader>wa - mark word as good
<leader>wr - mark word as bad

# Git

<leader>dpl - inline preview diff
<leader>dr - reset diff
<leader>dbv - check buffer diff
<leader>do - open diff file menu
<leader>dh - open diff history menu
<leader>dc - close diff menu
<leader>G - open git menu (s - stage, u - unstage, U - all unstage, = - view diff, cc - commit)

# Tabs

<leader>bp - pin buffer tab
<leader>bq - close all buffer except current
<A-,> - move tab left
<A-.> - move tab right

# Tags

<leader>tg - tagbar on the right side
<leader>at - tag navigation modal bar

# Telescope

<leader>ff - find files name
<leader>fg - find files content
<leader>fb - find files name in opened buffers
<leader>fq - find quick fix
<C-t> - open telescope result in buffer

# LSP

<leader>gd - go to definition
<leader>gb - back from last definition
<leader>gf - forward to last definition
<leader>gr - list of references
<leader>I - information
<leader>e - error description
<leader>rn - rename variable in project
<leader>fe - find current buffer errors
<leader>fea - find all errors
<leader>gne - goto next error
<leader>gpe - goto prev error

# Autoclose

ciw - rename tag
cs/[source/]/[target/] - replace symbols like quotes at start and end of current text

# Fold

zr - close all depth except first
zo - open all
zc - close all
z - toggle
Z - view short content

# Auto comment

gcc - toggle comment in normal mode
gc - toggle comment in visual mode
gbc - toggle comment block in normal mode
gb - toggle comment block in visual mode

# Fast move

s - to search
S - to treeSitter object

# Surround

ds\* - delete surround
cs\*\* - change surround from to
asw\* - add surround
cst - change both of tag
dst - delete tag without children
